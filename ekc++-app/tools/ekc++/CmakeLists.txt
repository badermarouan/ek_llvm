add_executable (ekc++)

target_link_libraries(ekc++ PRIVATE
    IRGen
    Bytecode
    Runtime
    LLVM
)

target_include_directories(ekc++
    PUBLIC
    ${PROJECT_SOURCE_DIR}/ekc++/include/
)

target_sources(ekc++
    PRIVATE
    ./sources/main.cpp
)

if(FROM_SOURCE)
    # Create a symlink to Clang's resource directory so ekc++ can find built-in headers
    # Clang's GetResourcesPath computes: <executable_dir>/../lib/clang/<version>
    # If ekc++ is at build/tools/ekc++/ekc++, it looks for build/tools/lib/clang/<version>
    # So we create: build/tools/lib/clang -> build/_deps/llvm-build/lib/clang
    add_custom_command(TARGET ekc++ POST_BUILD
        COMMAND ${CMAKE_COMMAND} -E make_directory "${CMAKE_BINARY_DIR}/tools/lib"
        COMMAND ${CMAKE_COMMAND} -E create_symlink
            "${llvm_BINARY_DIR}/lib/clang"
            "${CMAKE_BINARY_DIR}/tools/lib/clang"
        COMMENT "Creating symlink to Clang resource directory"
    )
endif()
